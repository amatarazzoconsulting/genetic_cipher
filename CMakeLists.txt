cmake_minimum_required(VERSION 3.16)
project(genetic_cipher LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Threads REQUIRED)
enable_testing()
add_executable(engine_tests tests/engine_tests.cpp)
target_link_libraries(engine_tests PRIVATE Threads::Threads)
add_test(NAME engine_tests COMMAND engine_tests)
add_executable(demo demo.cpp)
target_link_libraries(demo PRIVATE Threads::Threads)
add_test(NAME demo COMMAND demo)
add_executable(shape_tests tests/shape_tests.cpp)
target_link_libraries(shape_tests PRIVATE Threads::Threads)
add_test(NAME shape_tests COMMAND shape_tests)
option(SECURITY_COMPARISON "Build diagnostics against OpenSSL AEAD ciphers" OFF)
if(SECURITY_COMPARISON)
  find_package(OpenSSL REQUIRED)
  add_executable(security_comparison tests/security_comparison.cpp)
  target_link_libraries(security_comparison PRIVATE Threads::Threads OpenSSL::Crypto)
  add_test(NAME security_comparison COMMAND security_comparison)
endif()
option(ENABLE_SANITIZERS "Enable AddressSanitizer and UndefinedBehaviorSanitizer (GCC/Clang)" OFF)
if(ENABLE_SANITIZERS)
  if(NOT CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    message(FATAL_ERROR "ENABLE_SANITIZERS requires GCC or Clang")
  endif()
  foreach(test_target engine_tests shape_tests)
    target_compile_options(${test_target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(${test_target} PRIVATE -fsanitize=address,undefined)
  endforeach()
endif()
